   130→    totalIncome: 0,
   131→    totalExpense: 0,
   132→    balance: 0,

   135→    cashBalance: 0,
   136→    bankIncome: 0,
   137→    bankExpense: 0,
   138→    bankBalance: 0,
   139→    filteredIncome: 0,

   145→  // Filters

   182→    familyId: '',
   183→    type: 'RONDA',
   184→    amount: '',

   297→        description: txForm.description,
   298→        date: txForm.date,

   305→          type: 'INCOME',
   306→          category: 'IURAN_BULANAN',
   307→          account: 'CASH',
   308→          amount: '',
   309→          description: '',
   310→          date: new Date().toISOString().split('T')[0],
   311→        });
   312→        fetchTransactions();
   313→      } else {
   314→        const data = await res.json();
   315→        toast.error(data.error || 'Gagal menambahkan transaksi');
   316→      }
   317→    } catch {
   318→      toast.error('Gagal menambahkan transaksi');
   319→    } finally {
   320→      setTxSaving(false);
   321→    }
   322→  };
   323→
   324→  const handleDeleteTransaction = async (id: string) => {
   325→    try {
   326→      const res = await api.delete(`/transactions?id=${id}`);
   327→      if (res.ok) {
   328→        toast.success('Transaksi berhasil dihapus');
   329→        fetchTransactions();
   330→      } else {
   331→        toast.error('Gagal menghapus transaksi');
   332→      }
   333→    } catch {
   334→      toast.error('Gagal menghapus transaksi');
   335→    }
   336→  };
   337→

   339→    if (!transferForm.amount || !transferForm.description || !transferForm.date) return;
   340→    setTransferSaving(true);

   342→      const res = await api.post('/transactions', {
   343→        type: 'TRANSFER',
   344→        category: direction,
   345→        amount: parseInt(transferForm.amount, 10),
   346→        description: transferForm.description,
   347→        date: transferForm.date,
   348→      });
   349→      if (res.ok) {
   350→        toast.success(
   351→          direction === 'SETOR_BANK'
   352→            ? 'Setoran ke bank berhasil'
   353→            : 'Penarikan dari bank berhasil'

   356→          direction: 'SETOR_BANK',
   357→          amount: '',
   358→          description: '',
   359→          date: new Date().toISOString().split('T')[0],
   360→        });
   361→        fetchTransactions();
   362→      } else {
   363→        const data = await res.json();
   364→        toast.error(data.error || 'Gagal melakukan transfer');
   365→      }
   366→    } catch {
   367→      toast.error('Gagal melakukan transfer');
   368→    } finally {
   369→      setTransferSaving(false);
   370→    }
   371→  };
   372→
   373→  const handleCreateFine = async () => {
   374→    if (!fineForm.userId || !fineForm.familyId || !fineForm.type || !fineForm.amount || !fineForm.reason || !fineForm.date)
   375→      return;
   376→    setFineSaving(true);

   378→      const res = await api.post('/fines', {
   379→        userId: fineForm.userId,
   380→        familyId: fineForm.familyId,
   381→        type: fineForm.type,
   382→        amount: parseInt(fineForm.amount, 10),
   383→        reason: fineForm.reason,
   384→        date: fineForm.date,
   385→      });

   387→        toast.success('Denda berhasil ditambahkan');
   388→        setFineOpen(false);
   389→        setFineForm({
   390→          userId: '',
   391→          familyId: '',
   392→          type: 'RONDA',
   393→          amount: '',
   394→          reason: '',
   395→          date: new Date().toISOString().split('T')[0],
   396→        });
   397→        fetchFines();
   398→      }
   399→    } catch {
   400→      toast.error('Gagal menambahkan denda');
   401→    } finally {
   402→      setFineSaving(false);
   403→    }

   408→      const res = await api.put('/fines', { id, status: 'PAID' });
   409→      if (res.ok) {
   410→        toast.success('Denda ditandai lunas');

   421→      ...prev,

   447→    return { className: 'bg-red-100 text-red-800', label: 'Keluar' };

   455→      // SETOR_BANK from BANK_BKK account = money coming in (shown as +)

   526→                    }}

   528→                    <SelectTrigger className="h-10 w-full">
   529→                      <SelectValue />
   530→                    </SelectTrigger>
   531→                    <SelectContent>
   532→                      <SelectItem value="INCOME">Pemasukan</SelectItem>

   535→                  </Select>

   591→                    className="h-10"
   592→                  />
   593→                </div>
   594→
   595→                <div className="space-y-2">
   596→                  <Label>Keterangan</Label>
   597→                  <Input
   598→                    placeholder="Deskripsi transaksi"
   599→                    value={txForm.description}
   600→                    onChange={(e) => setTxForm({ ...txForm, description: e.target.value })}
   601→                    className="h-10"
   602→                  />
   603→                </div>
   604→
